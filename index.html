import turtle as t

t.speed(0)
for size in range(120, 40, -20):
    for _ in range(5):
        t.forward(size)
        t.right(72)
    t.penup()

Signal rings 
import turtle as t

t.speed(0)
for radius in range(20, 140, 10):
    t.penup()

    t.pendown()
